📝 Функция add_note() — создание заметки

def add_note():
    note_name, ok = QInputDialog.getText(notes_win, "Добавить заметку", "Название: ")
    if ok and note_name != "":
        notes[note_name] = {"текст" : "", "теги" : []}
        list_notes.addItem(note_name)
        list_tags.addItems(notes[note_name]["теги"])
        print(notes)

💾 Функция save_note() — сохранение заметки

def save_note():
    if list_notes.selectedItems():
        key = list_notes.selectedItems()[0].text()
        notes[key]["текст"] = field_text.toPlainText()
        with open("notes_data.json", "w") as file:
            json.dump(notes, file, sort_keys=True, ensure_ascii=False)
        print(notes)
    else:
        print("Заметка для сохранения не выбрана!")

🗑️ Функция del_note() — удаление заметки

def del_note():
    if list_notes.selectedItems():
        key = list_notes.selectedItems()[0].text()
        del notes[key]
        list_notes.clear()
        list_tags.clear()
        field_text.clear()
        list_notes.addItems(notes)
        with open("notes_data.json", "w") as file:
            json.dump(notes, file, sort_keys=True, ensure_ascii=False)
        print(notes)
    else:
        print("Заметка для удаления не выбрана!")

🏷️ Функция add_tag() — добавление тега

def add_tag():
    if list_notes.selectedItems():
        key = list_notes.selectedItems()[0].text()
        tag = field_tag.text()
        if not tag in notes[key]["теги"]:
            notes[key]["теги"].append(tag)
            list_tags.addItem(tag)
            field_tag.clear()
        with open("notes_data.json", "w") as file:
            json.dump(notes, file, sort_keys=True, ensure_ascii=False)
        print(notes)
    else:
        print("Заметка для добавления тега не выбрана!")
Шаг 1

Проверяет, выбрана ли заметка

Шаг 2

Получает название заметки и текст тега из поля ввода

Шаг 3

Проверяет, нет ли уже такого тега (защита от дубликатов)

Шаг 4

Добавляет тег в словарь и в список, очищает поле ввода

Шаг 5

Сохраняет изменения в JSON файл

Особенность: Проверяет уникальность тега перед добавлением

❌ Функция del_tag() — удаление тега

def del_tag():
    if list_tags.selectedItems():
        key = list_notes.selectedItems()[0].text()
        tag = list_tags.selectedItems()[0].text()
        notes[key]["теги"].remove(tag)
        list_tags.clear()
        list_tags.addItems(notes[key]["теги"])
        with open("notes_data.json", "w") as file:
            json.dump(notes, file, sort_keys=True, ensure_ascii=False)
    else:
        print("Тег для удаления не выбран!")
Шаг 1

Проверяет, выбран ли тег в списке тегов (не заметок!)

Шаг 2

Получает название заметки и текст выбранного тега

Шаг 3

Удаляет тег из списка тегов заметки в словаре

Шаг 4

Перестраивает список тегов и сохраняет в файл

Особенность: Работает с двумя списками — list_notes и list_tags, проверяет выбор в list_tags

🔍 Функция search_tag() — поиск по тегу

def search_tag():
    tag = field_tag.text()
    if button_tag_search.text() == "Искать заметки по тегу" and tag:
        notes_filtered = {}
        for note in notes:
            if tag in notes[note]["теги"]: 
                notes_filtered[note] = notes[note]
        button_tag_search.setText("Сбросить поиск")
        list_notes.clear()
        list_tags.clear()
        list_notes.addItems(notes_filtered)
    elif button_tag_search.text() == "Сбросить поиск":
        field_tag.clear()
        list_notes.clear()
        list_tags.clear()
        list_notes.addItems(notes)
        button_tag_search.setText("Искать заметки по тегу")
Режим 1

Поиск

Создаёт новый словарь notes_filtered с заметками, у которых есть искомый тег

Режим 2

Сброс

Восстанавливает полный список всех заметок

Особенность: Использует переключатель — меняет текст кнопки для переключения между режимами поиска и сброса
